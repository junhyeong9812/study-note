/** 맨해튼 거리. 상하좌우로만 갈 때의 최소 걸음 수다. */

public static final ToIntBiFunction<int[], int[]> INFLATED =
        (from, to) -> 3 * (Math.abs(from[0] - to[0]) + Math.abs(from[1] - to[1]));
